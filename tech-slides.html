Certified View: Student Momentum

Census-aligned momentum flag (15-credit threshold example). 

CREATE OR ALTER VIEW dbo.vw_student_momentum AS
SELECT
  student_id,
  term_code,
  program_code,
  credits_attempted,
  CASE WHEN credits_attempted >= 15 THEN 1 ELSE 0 END AS momentum_15_flag,
  CASE WHEN retained_next_term = 1 THEN 1 ELSE 0 END AS persisted_flag
FROM dbo.banner_enrollment
WHERE census_flag = 1;
GO

Integrated Program Summary

Aggregates Banner + Navigate + Ad Astra into executive-ready metrics. 

SELECT
  e.term_code,
  e.program_code,
  COUNT(DISTINCT e.student_id) AS headcount,
  COUNT(DISTINCT CASE WHEN r.risk_level IN ('High','Medium')
       THEN e.student_id END) AS at_risk,
  ROUND(
    CAST(COUNT(DISTINCT CASE WHEN r.risk_level IN ('High','Medium')
       THEN e.student_id END) AS FLOAT)
    / NULLIF(COUNT(DISTINCT e.student_id),0) * 100, 2
  ) AS at_risk_rate_pct
FROM banner_enrollment e
LEFT JOIN navigate_risk_signals r
  ON r.student_id = e.student_id
 AND r.term_code = e.term_code
WHERE e.census_flag = 1
GROUP BY e.term_code, e.program_code;

Validation Controls

Prevents inflated risk or retention metrics due to orphan records. 

SELECT r.student_id, r.term_code
FROM navigate_risk_signals r
LEFT JOIN banner_enrollment e
  ON e.student_id = r.student_id
 AND e.term_code = r.term_code
WHERE e.student_id IS NULL;
